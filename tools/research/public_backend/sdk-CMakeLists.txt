cmake_minimum_required(VERSION 3.18)
project(rnp_only_sdk_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FETCHCONTENT_SOURCE_DIR_RNP "${CMAKE_CURRENT_SOURCE_DIR}/rnp" CACHE PATH "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(DOWNLOAD_GTEST OFF CACHE BOOL "" FORCE)
set(ENABLE_DOC Off CACHE STRING "" FORCE)
set(CRYPTO_BACKEND openssl CACHE STRING "" FORCE)
set(SYSTEM_LIBSEXPP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LIBPGPFACTORY_RNP_ONLY ON CACHE BOOL "" FORCE)
add_subdirectory(factory/libpgpfactory factory-build EXCLUDE_FROM_ALL)
add_executable(public_backend_probe probe.cpp)
target_link_libraries(public_backend_probe PRIVATE libpgpfactory)
