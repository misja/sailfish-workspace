cmake_minimum_required(VERSION 3.18)
project(public_backend_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GPGFACTORY_SOURCE_DIR "" CACHE PATH "Verified e53cd22c source tree")
set(RNP_SOURCE_DIR "" CACHE PATH "Verified aad1892e RNP source tree")
if(NOT EXISTS "${GPGFACTORY_SOURCE_DIR}/libpgpfactory/export/GpgIdManage.h"
   OR NOT EXISTS "${RNP_SOURCE_DIR}/cmake/version.cmake")
  message(FATAL_ERROR "Supply both complete, independently hash-verified pinned source trees")
endif()
# Upstream FetchContent's local-source override prevents its mutable tag fetch.
set(FETCHCONTENT_SOURCE_DIR_RNP "${RNP_SOURCE_DIR}" CACHE PATH "Pinned local RNP" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No automatic downloads" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "No automatic updates" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "No unrelated upstream tests" FORCE)
set(DOWNLOAD_GTEST OFF CACHE BOOL "No GTest downloads" FORCE)
set(ENABLE_DOC Off CACHE STRING "No documentation build" FORCE)
set(SYSTEM_LIBSEXPP ON CACHE BOOL "Upstream-supported system sexpp>=0.8.7" FORCE)
set(CRYPTO_BACKEND botan3 CACHE STRING "Retained Botan3 backend" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build-tree libraries; no install" FORCE)
add_subdirectory("${GPGFACTORY_SOURCE_DIR}/libpgpfactory" pgpfactory EXCLUDE_FROM_ALL)
add_executable(public_backend_probe probe.cpp)
target_link_libraries(public_backend_probe PRIVATE libpgpfactory)
